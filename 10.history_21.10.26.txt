Текущее состояние: В app/templates/ лежат правильные файлы (я их сверил — они идентичны тем, что захардкожены). Но create_agent в main_window.py их игнорирует и пишет строки из кода.

def parse_interview_file(file_path: Path) -> Dict[str, str]:
    import re
    with open(file_path, "r", encoding="utf-8") as f:
        content = f.read()
    section_pattern = re.compile(
        r"##\s*(\d+)\.\s*([А-Я_]+)\s*\n(.*?)(?=\n##\s*\d+\.|$)",
        re.DOTALL | re.IGNORECASE,
    )
    result = {}
    matches = section_pattern.findall(content)
    for num, name, text in matches:
        clean_text = text.strip()
        clean_text = re.sub(r"^[\s\-*0-9.]+\s*", "", clean_text, flags=re.MULTILINE)
        result[name] = clean_text
    return result


def parse_interviews_folder(folder_path: Path, markers: List[str]) -> Dict[str, str]:
    combined_data = {}
    extensions = [".txt", ".md", ".docx"]
    for ext in extensions:
        for file_path in folder_path.glob(f"*{ext}"):
            logger.info(f"Обрабатываю: {file_path.name}")
            try:
                if ext == ".docx":

def parse_interview_from_text(content: str, markers: List[str]) -> Dict[str, str]:
    result = {}
    for marker in markers:
        pattern = re.compile(
            rf"##\s*\d+\.\s*{marker}\s*\n(.*?)(?=\n##\s*\d+\.|$)",
            re.DOTALL | re.IGNORECASE,
        )
        match = pattern.search(content)
        if match:
            text = match.group(1).strip()
            text = re.sub(r"^[\s\-*0-9.]+\s*", "", text, flags=re.MULTILINE)
            result[marker] = text
        else:
            simple_pattern = re.compile(
                rf"{marker}\s*\n(.*?)(?=\n\s*[А-Я_]+|$)", re.DOTALL | re.IGNORECASE
            )
            simple_match = simple_pattern.search(content)
            if simple_match:
                text = simple_match.group(1).strip()
                result[marker] = text
            else:
                result[marker] = ""
    return result

def extract_json_from_text(text: str) -> dict | None:
    if not text:
        return None
    code_block_match = re.search(r"```json\s*([\s\S]*?)\s*```", text, re.DOTALL)
    if code_block_match:
        json_str = code_block_match.group(1).strip()
        try:
            return json.loads(json_str)
        except json.JSONDecodeError:
            pass
    json_match = re.search(
        r'\{[^{}]*"tool"\s*:\s*"[^"]+"\s*,\s*"params"\s*:\s*\{[^{}]*\}\s*\}',
        text,
        re.DOTALL,
    )
    if json_match:
        try:
            return json.loads(json_match.group(0))
        except json.JSONDecodeError:
            pass
    try:
        start = text.find("{")
        if start != -1:
            depth = 0
            for i, char in enumerate(text[start:], start):
                if char == "{":
                    depth += 1
                elif char == "}":
                    depth -= 1
                    if depth == 0:
                        potential_json = text[start : i + 1]
                        try:
                            parsed = json.loads(potential_json)
                            if isinstance(parsed, dict):
                                return parsed
                        except json.JSONDecodeError:
                            pass
                        break
    except Exception:
        pass
    try:
        return json.loads(text.strip())
    except json.JSONDecodeError:
        pass
    try:
        fixed_text = re.sub(r",\s*\}", "}", text)
        fixed_text = re.sub(r",\s*\]", "]", fixed_text)
        fixed_text = re.sub(r",\s*,", ",", fixed_text)
        start = fixed_text.find("{")
        if start != -1:
            depth = 0
            for i, char in enumerate(fixed_text[start:], start):
                if char == "{":
                    depth += 1
                elif char == "}":
                    depth -= 1
                    if depth == 0:
                        potential_json = fixed_text[start : i + 1]
                        try:
                            parsed = json.loads(potential_json)
                            if isinstance(parsed, dict):
                                return parsed
                        except json.JSONDecodeError:
                            pass
                        break
    except Exception:
        pass
    return None

# ==================== ПАРСЕР ЗАПРОСОВ НА СОЗДАНИЕ ФАЙЛОВ ====================

def parse_file_request(
    message: str,
    project_name: str = DEFAULT_PROJECT_NAME,
    block_name: str = DEFAULT_BLOCK_NAME,
) -> dict | None:
    msg_lower = message.lower()
    logger.debug(f"Парсер обрабатывает: {message[:50]}...")

    if message.startswith("[SYSTEM_FOLLOWUP]"):
        logger.debug("FOLLOW-UP — пропускаем парсер")
        return None

    if re.search(r"[A-Za-z]:[\\\/]", message):
        logger.debug("В запросе есть путь к файлу — передаём LLM")
        return None

    file_keywords = [
        "создай", "сделай", "сгенерируй", "word", "ворд", "docx", "doc",
        "excel", "ексел", "xlsx", "xls", "ppt", "pptx", "презентац",
        "xml", "таблиц", "отчёт", "документ",
    ]

    if any(kw in msg_lower for kw in ["прочитай файлы", "покажи содержимое папки", "прочитай папку"]):
        logger.info("Распознана команда чтения папки")
        folder_match = re.search(r"папк[иа]\s+([A-Za-z]:[\\\/][^\s,;.]+)", message, re.IGNORECASE)
        if not folder_match:
            folder_match = re.search(r'["\']([^"\']+)["\']', message)
        folder_path = folder_match.group(1) if folder_match else ""
        return {"tool": "read_files_from_folder", "params": {"folder_path": folder_path}}

    if not any(kw in msg_lower for kw in file_keywords):
        return None

    logger.info("Похоже на запрос файла")

    study_keywords = ["изучи", "прочитай", "ознакомься", "изучить", "прочитать"]
    methodology_keywords = ["стандарт", "методологи", "гост", "pmbok", "prince", "методик"]
    if any(kw in msg_lower for kw in study_keywords) and any(kw in msg_lower for kw in methodology_keywords):
        logger.info("Распознана команда изучения методологии")
        file_match = re.search(r"([A-Za-z]:[\\\/][^\s,;]+\.(docx|pdf|txt|md))", message, re.IGNORECASE)
        if file_match:
            file_path = file_match.group(1)
            return {"tool": "study_methodology", "params": {"file_path": file_path}}
        return None

    remember_keywords = ["запомни", "запомнить", "сохрани в память", "прочти и запомни"]
    if any(kw in msg_lower for kw in remember_keywords):
        logger.info("Распознана команда запоминания файла")
        file_match = re.search(r"([A-Za-z]:[\\\/][^\s,;.]+)", message)
        if file_match:
            file_path = file_match.group(1)
            return {"tool": "remember_file", "params": {"file_path": file_path}}
        return None

    project_match = re.search(r"проект\s*[::]\s*([^,;.]+)", message, re.IGNORECASE)
    block_match = re.search(r"блок\s*[::]\s*([^,;.]+)", message, re.IGNORECASE)
    final_project_name = project_match.group(1).strip() if project_match else project_name
    final_block_name = block_match.group(1).strip() if block_match else block_name

    if any(kw in msg_lower for kw in ["сделай тз", "создай тз", "тз из данных"]):
        logger.info("Распознана команда создания ТЗ из данных")
        folder_path = ""
        folder_match = re.search(r"папк[иеи]\s+([A-Za-z]:[\\\/][^\s,;.]+)", message, re.IGNORECASE)
        if not folder_match:
            folder_match = re.search(r'["\']([^"\']+)["\']', message)
        if not folder_match:
            folder_match = re.search(r"([A-Za-z]:[\\\/][^\s,;.]+)", message)
        if folder_match:
            folder_path = folder_match.group(1)

        template_path = None
        template_match = re.search(r"шаблон[уе]?\s+([A-Za-z]:[\\\/][^\s,;]+)", message, re.IGNORECASE)
        if not template_match:
            template_match = re.search(r"по\s+шаблон[уе]?\s+([A-Za-z]:[\\\/][^\s,;]+)", message, re.IGNORECASE)
        if not template_match:
            template_match = re.search(r'шаблон[уе]?\s+["\']([^"\']+)["\']', message, re.IGNORECASE)
        if template_match:
            template_path = template_match.group(1)

        return {
            "tool": "create_doc_from_folder",
            "params": {
                "folder_path": folder_path,
                "project_name": final_project_name,
                "block_name": final_block_name,
                "template_path": template_path,
                "filename": "ТЗ_из_данных",
            },
        }

    if any(kw in msg_lower for kw in ["excel", "xlsx", "ексел", "таблиц"]):
        logger.info("Распознан Excel")
        headers_text = ""
        rows_text = ""
        headers_pos = re.search(r"заголовки?\s*[::]", message, re.IGNORECASE)
        rows_pos = re.search(r"строки?\s*[::]", message, re.IGNORECASE)
        if headers_pos:
            start = headers_pos.end()
            if rows_pos and rows_pos.start() > start:
                headers_text = message[start:rows_pos.start()].strip()
            else:
                headers_text = message[start:].strip()
        if rows_pos:
            rows_text = message[rows_pos.end():].strip()
        headers = [h.strip() for h in headers_text.split(",")] if headers_text else ["Данные"]
        rows = []
        if rows_text:
            row_parts = [r.strip() for r in rows_text.split(";") if r.strip()]
            for row in row_parts:
                cells = [c.strip() for c in row.split(",")]
                while len(cells) < len(headers):
                    cells.append("")
                rows.append(cells[:len(headers)])
        else:
            rows = [[""] * len(headers)]
        return {
            "tool": "create_xlsx",
            "params": {
                "project_name": final_project_name,
                "block_name": final_block_name,
                "filename": "Таблица_данных",
                "data": {"headers": headers, "rows": rows},
            },
        }

    if any(kw in msg_lower for kw in ["word", "docx", "отчёт", "ворд", "документ"]):
        content_match = re.search(r"содержание\s*[::]\s*(.+?)(?=\.\s*$|$)", message, re.IGNORECASE)
        content = content_match.group(1).strip() if content_match else message
        return {
            "tool": "create_docx",
            "params": {
                "project_name": final_project_name,
                "block_name": final_block_name,
                "filename": "Документ",
                "content": content,
            },
        }

    if any(kw in msg_lower for kw in ["ppt", "pptx", "презентац"]):
        logger.info("Распознана презентация")
        slides = []
        slide_pattern = re.compile(
            r"(?:первый|второй|третий|четвёртый|пятый|\d+)\s*слайд\s*[::]\s*([^,;.]+?)\s*,\s*текст\s*[::]\s*([^,;.]+)",
            re.IGNORECASE,
        )
        matches = slide_pattern.findall(message)
        if matches:
            slides = [{"title": t.strip(), "content": c.strip()} for t, c in matches]
        else:
            for i in range(1, 10):
                pattern = re.compile(rf"{i}\s*слайд\s*[::]\s*([^,;.]+?)\s*,\s*текст\s*[::]\s*([^,;.]+)", re.IGNORECASE)
                match = pattern.search(message)
                if match:
                    slides.append({"title": match.group(1).strip(), "content": match.group(2).strip()})
                else:
                    pattern2 = re.compile(rf"{i}\s*слайд\s*[::]\s*([^,;.]+?)\s*[::]\s*([^,;.]+)", re.IGNORECASE)
                    match2 = pattern2.search(message)
                    if match2:
                        slides.append({"title": match2.group(1).strip(), "content": match2.group(2).strip()})
        if not slides:
            slides = [{"title": "Слайд 1", "content": "Содержание презентации"}]
        return {
            "tool": "create_pptx",
            "params": {
                "project_name": final_project_name,
                "block_name": final_block_name,
                "filename": "Презентация",
                "slides": slides,
            },
        }

    if "xml" in msg_lower:
        data = {}
        data_matches = re.findall(r"(\w+)\s*[::]\s*([^,;.]+)", message, re.IGNORECASE)
        for key, value in data_matches:
            key = key.strip().lower()
            if key not in ["проект", "блок", "файл", "создай", "сделай", "заголовки", "строки", "содержание"]:
                data[key] = value.strip()
        if not data:
            data = {"message": "Данные для XML"}
        return {
            "tool": "create_xml",
            "params": {
                "project_name": final_project_name,
                "block_name": final_block_name,
                "filename": "data",
                "root_tag": "root",
                "data": data,
            },
        }

    return None

@app.post("/agents/{agent_id}/chat")
async def chat_with_agent(agent_id: int, query: AgentQuery):
    db = get_db()
    if db is None:
        return {"error": "База данных не инициализирована. Укажите каталог данных в настройках."}
    with db as session:
        agent = session.query(AgentDB).filter(AgentDB.id == agent_id).first()
        if not agent:
            return {"error": "Агент не найден"}

        project_name = DEFAULT_PROJECT_NAME
        block_name = DEFAULT_BLOCK_NAME
        if query.project_id:
            project = session.query(ProjectDB).filter(ProjectDB.id == query.project_id).first()
            if project:
                project_name = project.name
        if query.block_id:
            block = session.query(BlockDB).filter(BlockDB.id == query.block_id).first()
            if block:
                block_name = block.name

        if agent.skill_id:
            skill = session.query(SkillDB).filter(SkillDB.id == agent.skill_id).first()
            system_prompt = skill.system_prompt if skill else agent.system_prompt
        else:
            system_prompt = agent.system_prompt or "Ты — полезный ИИ-ассистент."

        parsed_command = parse_file_request(query.message, project_name, block_name)

        if parsed_command:
            json_response = json.dumps(parsed_command, ensure_ascii=False)
            logger.info(f"ПАРСИНГ: распознан как {parsed_command['tool']}")
            
            user_msg = MessageDB(
                task_id=query.task_id, session_id=f"task_{query.task_id}",
                role="user", content=query.message, agent_id=None,
                created_at=datetime.now(timezone.utc),
            )
            session.add(user_msg)
            session.commit()
            agent_msg = MessageDB(
                task_id=query.task_id, session_id=f"task_{query.task_id}",
                role="agent", agent_id=agent.id, content=json_response,
                created_at=datetime.now(timezone.utc),
            )
            session.add(agent_msg)
            session.commit()
            return {
                "agent_id": agent.id, "agent_name": agent.name,
                "response": json_response, "system_prompt_used": system_prompt,
                "task_id": query.task_id, "project_name": project_name,
                "block_name": block_name,
            }

        user_msg = MessageDB(
            task_id=query.task_id, session_id=f"task_{query.task_id}",
            role="user", content=query.message, agent_id=None,
            created_at=datetime.now(timezone.utc),
        )
        session.add(user_msg)
        session.commit()

        history = (
            session.query(MessageDB).filter(MessageDB.task_id == query.task_id)
            .order_by(MessageDB.created_at).limit(10).all()
        )
        messages_for_llm = []
        for msg in history:
            if msg.role == "user":
                messages_for_llm.append({"role": "user", "content": msg.content})
            elif msg.role == "agent":
                messages_for_llm.append({"role": "assistant", "content": msg.content})

        context_message = f"""
Текущий контекст:
- Проект: {project_name}
- Блок: {block_name}
- ID задачи: {query.task_id}

Запрос пользователя: {query.message}

Если пользователь просит создать файл — используй project_name и block_name из контекста.
"""

        try:
            max_tokens = int(keyring.get_password("DigitalWorker", "max_tokens") or "2000")
        except:
            max_tokens = 2000
        try:
            temperature = float(keyring.get_password("DigitalWorker", "temperature") or "0.7")
        except:
            temperature = 0.7

        llm_payload = {
            "messages": [
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": context_message},
            ],
            "temperature": temperature,
            "max_tokens": max_tokens,
        }

        llm_config = get_active_llm_config()
        if not llm_config:
            return {"error": "LLM не настроена. Добавьте провайдера через Параметры → Управление провайдерами."}
        llm_url = llm_config.get("base_url")
        llm_model = llm_config.get("model")
        llm_type = llm_config.get("type")
        if not llm_url:
            return {"error": "Невозможно отработать запрос. Укажите адрес LLM в настройках."}
        if not llm_model:
            return {"error": "Невозможно отработать запрос. Укажите модель LLM в настройках."}

        logger.info(f"Используется LLM: {llm_type} ({llm_model})")

        headers = {"Content-Type": "application/json"}
        if llm_type == "cloud" and llm_config.get("api_key"):
            headers["Authorization"] = f"Bearer {llm_config.get('api_key')}"
        llm_payload["model"] = llm_model

        try:
            response = requests.post(
                f"{llm_url}/chat/completions",
                json=llm_payload,
                headers=headers,
                timeout=TIMEOUT_LLM_REQUEST,
            )
            if response.status_code == 200:
                data = response.json()
                llm_answer = data["choices"][0]["message"]["content"]
                
                logger.info("Получен ответ от LLM")
                
                agent_msg = MessageDB(
                    task_id=query.task_id, session_id=f"task_{query.task_id}",
                    role="agent", agent_id=agent.id, content=llm_answer,
                    created_at=datetime.now(timezone.utc),
                )
                session.add(agent_msg)
                session.commit()
                return {
                    "agent_id": agent.id, "agent_name": agent.name,
                    "response": llm_answer, "system_prompt_used": system_prompt,
                    "task_id": query.task_id, "project_name": project_name,
                    "block_name": block_name,
                }
            else:
                error_msg = f"Ошибка LLM: {response.status_code}, {response.text}"
                logger.error(error_msg)
                return {"error": error_msg}
        except Exception as e:
            error_msg = f"Ошибка подключения к LLM: {e}"
            logger.error(error_msg)

        template_path = request.get("template_path")

        template_path = Path(template_path)
        if not folder.exists():
            return {"status": "error", "message": f"Папка не найдена: {folder_path}"}
        if not template_path.exists():
            return {"status": "error", "message": f"Шаблон не найден: {template_path}"}

        markers = file_service.analyze_template_structure(template_path)
        if not markers:
            return {"status": "error", "message": "В шаблоне не найдено маркеров. Используйте формат [НАЗВАНИЕ_РАЗДЕЛА]."}

        interview_data = parse_interviews_folder(folder, markers)
        from app.config import SKILLS_ROOT
        prompt_path = Path(SKILLS_ROOT) / "Аналитик_1С" / "extract_prompt.md"
        if not prompt_path.exists():
            return {"status": "error", "message": f"Файл с промптом не найден: {prompt_path}"}
        with open(prompt_path, "r", encoding="utf-8") as f:
            base_prompt = f.read()

        markers_list = "\n".join([f"- {m}" for m in markers])
        interview_text = json.dumps(interview_data, ensure_ascii=False, indent=2)
        extraction_prompt = f"""{base_prompt}

**Разделы для извлечения:**
{markers_list}

**Текст интервью:**
{interview_text[:8000]}
"""

        llm_config = get_active_llm_config()
        llm_url = llm_config.get("base_url")
        if not llm_url:
            return {"status": "error", "message": "LLM-сервер не настроен"}
        
        logger.info(f"Используется LLM: {llm_config.get('type')} ({llm_config.get('model')})")

        llm_payload = {
            "messages": [
                {"role": "system", "content": "Ты — ИИ-аналитик. Извлекай структурированные данные из текста."},
                {"role": "user", "content": extraction_prompt},

            "temperature": 0.3,
            "max_tokens": 4000,
            "model": llm_config.get("model")
        }
        headers = {"Content-Type": "application/json"}
        if llm_config.get("type") == "cloud" and llm_config.get("api_key"):
            headers["Authorization"] = f"Bearer {llm_config.get('api_key')}"

        llm_response = requests.post(f"{llm_url}/chat/completions", json=llm_payload, headers=headers, timeout=600)
        if llm_response.status_code != 200:
            return {"status": "error", "message": f"Ошибка LLM: {llm_response.status_code}"}
        llm_data = llm_response.json()
        extracted_text = llm_data["choices"][0]["message"]["content"]

        json_match = re.search(r"\{.*\}", extracted_text, re.DOTALL)
        if not json_match:
            return {"status": "error", "message": "LLM не вернула корректный JSON"}
        try:
            extracted_data = json.loads(json_match.group(0))
        except json.JSONDecodeError:
            return {"status": "error", "message": "Ошибка парсинга JSON из ответа LLM"}

        folder = file_service._get_project_block_path(project_name, block_name)
        output_path = folder / f"{filename}.docx"
        result_path = file_service.fill_template_from_docx(
            template_path=template_path, output_path=output_path, data=extracted_data
        )
        return {"status": "ok", "filepath": result_path, "extracted_data": extracted_data, "message": f"Файл создан: {result_path}"}

@app.post("/mcp/remember_file")
async def remember_file(request: dict):
    import threading
    file_path = request.get("file_path")
    agent_name = request.get("agent_name")
    task_id = request.get("task_id")
    if not file_path:
        return {"status": "error", "message": "Не указан путь к файлу"}
    if not agent_name:
        return {"status": "error", "message": "Не указано имя агента"}
    path = Path(file_path)
    if not path.exists():
        return {"status": "error", "message": f"Файл не найден: {file_path}"}
    if memory_manager is None:
        return {"status": "error", "message": "Менеджер памяти не инициализирован"}
    try:
        file_id = memory_manager.queue_file_for_indexing(agent_name, path)
        def background_summary():
            import time
            time.sleep(5)
            try:
                create_summary_for_file(file_id, agent_name, path, task_id)
            except Exception as e:
                logger.error(f"Ошибка создания выжимки: {e}")
        threading.Thread(target=background_summary, daemon=True).start()
        return {"status": "ok", "message": f"Файл '{path.name}' принят на индексацию (ID: {file_id})", "file_id": file_id, "filename": path.name, "async": True}
    except Exception as e:
        return {"status": "error", "message": str(e)}


def create_summary_for_file(file_id: int, agent_name: str, file_path, task_id: int = None):
    import sqlite3
    from pathlib import Path
    if memory_manager is None:
        return
    stored_path = memory_manager.skills_root / agent_name / "knowledge" / Path(file_path).name
    path = stored_path
    try:
        content = memory_manager._read_file_content(path)
    except Exception as e:
        logger.error(f"Не удалось прочитать файл {path.name}: {e}")
        return
    if not content.strip():
        return
    llm_config = get_active_llm_config()
    if not llm_config:
        return
    llm_url = llm_config.get("base_url")
    if not llm_url:
        return
    summary_prompt = f"""Прочитай следующий текст и сделай краткую выжимку (5-10 строк) о чём этот файл.
Ответ должен быть только текстом, без JSON, без Markdown-блоков.
Текст:
{content[:10000]}
Выжимка:"""
    llm_payload = {
        "messages": [
            {"role": "system", "content": "Ты — ассистент, который делает краткие выжимки текстов."},
            {"role": "user", "content": summary_prompt},
        ],
        "temperature": 0.3,
        "max_tokens": 500,
        "model": llm_config.get("model")
    }
    headers = {"Content-Type": "application/json"}
    if llm_config.get("type") == "cloud" and llm_config.get("api_key"):
        headers["Authorization"] = f"Bearer {llm_config.get('api_key')}"
    try:
        response = requests.post(f"{llm_url}/chat/completions", json=llm_payload, headers=headers, timeout=TIMEOUT_LLM_SUMMARY)
        if response.status_code == 200:
            data = response.json()
            summary = data["choices"][0]["message"]["content"].strip()
            logger.info(f"Выжимка для '{path.name}' создана ({len(summary)} символов)")
        else:
            logger.warning(f"Ошибка LLM при создании выжимки: {response.status_code}")
            return
    except Exception as e:
        logger.error(f"Ошибка вызова LLM для выжимки: {e}")
        return
    summary_path = path.parent / f"{path.name}.summary.md"
    try:
        with open(summary_path, "w", encoding="utf-8") as f:
            f.write(f"# Выжимка: {path.name}\n\n{summary}\n")
    except Exception as e:
        logger.error(f"Не удалось сохранить .summary.md: {e}")
        return
    try:
        conn = sqlite3.connect(memory_manager.db_path)
        cursor = conn.cursor()
        cursor.execute("UPDATE memory_files SET summary_path = ? WHERE id = ?", (str(summary_path), file_id))
        conn.commit()
        conn.close()
    except Exception as e:
        logger.error(f"Не удалось обновить БД: {e}")
    if task_id:
        try:
            db = get_db()
            if db:
                with db as session:
                    system_msg = MessageDB(
                        task_id=task_id, session_id=f"task_{task_id}",
                        role="system", agent_id=None,
                        content=f"📚 Запомнен файл: {path.name}\n\nЧто внутри:\n{summary}",
                        created_at=datetime.now(timezone.utc),
                    )
                    session.add(system_msg)
                    session.commit()
        except Exception as e:
            logger.error(f"Не удалось добавить системное сообщение: {e}")

@app.post("/methodology/study")
async def study_methodology(request: dict):
    file_path = request.get("file_path")
    if not file_path:
        return {"status": "error", "message": "Не указан путь к файлу"}
    path = Path(file_path)
    if not path.exists():
        return {"status": "error", "message": f"Файл не найден: {file_path}"}
    if path.is_dir():
        for ext in ["*.docx", "*.pdf", "*.txt", "*.md"]:
            files = list(path.glob(ext))
            if files:
                path = files[0]
                break
        if path.is_dir():
            return {"status": "error", "message": f"В папке {file_path} не найдено файлов методологии"}
    content = ""
    if path.suffix.lower() == ".pdf":
        try:
            import pypdf
            reader = pypdf.PdfReader(path)
            for page in reader.pages:
                content += page.extract_text() + "\n"
        except ImportError:
            return {"status": "error", "message": "Библиотека pypdf не установлена"}
        except Exception as e:
            return {"status": "error", "message": f"Ошибка чтения PDF: {e}"}
    elif path.suffix.lower() == ".docx":
        try:
            from docx import Document
            doc = Document(path)
            for para in doc.paragraphs:
                content += para.text + "\n"
        except ImportError:
            return {"status": "error", "message": "Библиотека python-docx не установлена"}
        except Exception as e:
            return {"status": "error", "message": f"Ошибка чтения DOCX: {e}"}
    elif path.suffix.lower() in [".txt", ".md"]:
        try:
            with open(path, "r", encoding="utf-8") as f:
                content = f.read()
        except Exception as e:
            return {"status": "error", "message": f"Ошибка чтения файла: {e}"}
    else:
        return {"status": "error", "message": f"Неподдерживаемый формат: {path.suffix}"}
    if not content.strip():
        return {"status": "error", "message": "Файл пуст или не содержит текста"}
    structure = extract_structure_from_text(content)
    return {
        "status": "ok", "filename": path.name,
        "structure": structure,
        "content_preview": content[:500] + ("..." if len(content) > 500 else ""),
        "total_chars": len(content),
    }


def extract_structure_from_text(text: str) -> List[str]:
    headers = []
    md_headers = re.findall(r"^(#{1,6})\s+(.+)$", text, re.MULTILINE)
    for level, title in md_headers:
        headers.append(title.strip())
    num_headers = re.findall(r"^(\d+\.\d+|\d+\.)\s+(.+)$", text, re.MULTILINE)
    for num, title in num_headers:
        if title.strip() not in headers:
            headers.append(title.strip())
    keyword_headers = re.findall(r"^(Глава|Раздел|Часть|Приложение)\s+(\d+\.?\d*)\s*[.:]?\s*(.+)$", text, re.MULTILINE | re.IGNORECASE)
    for keyword, num, title in keyword_headers:
        full_title = f"{keyword} {num}. {title.strip()}"
        if full_title not in headers:
            headers.append(full_title)
    if not headers:
        common_sections = ["Введение", "Объект автоматизации", "Цель", "Задачи", "Требования", "Функциональные требования", "Технические требования", "Состав работ", "Методология", "Алгоритмы", "Отчетность", "Контроль", "Поддержка", "Заключение", "Список литературы"]
        found_sections = [s for s in common_sections if re.search(rf"\b{s}\b", text, re.IGNORECASE)]
        if found_sections:
            headers = found_sections
    unique_headers = []
    seen = set()
    for h in headers:
        if h.lower() not in seen:
            seen.add(h.lower())
            unique_headers.append(h)
    return unique_headers

⚠️ Проблемы (честно)
Проблема 1: Порядок вызовов — это не просто список инструментов
Ты пишешь: «Презентация → read_file, потом create_pptx».
Но LLM не следует этой последовательности буквально. Смотри лог:
text
LLM вернула ответ: 'В файле «Память.txt» был такой текст: ...'  ← ПЕРЕСКАЗ, а не create_pptx
Причина: LLM считает, что после read_file задача выполнена. Она не видит явной команды «продолжай до create_pptx».
Решение: в agents.md написать не просто «делай так», а жёстко: «После read_file — ОБЯЗАТЕЛЬНО create_pptx. НЕ ОСТАНАВЛИВАЙСЯ после read_file».
Проблема 2: Куда попадает agents.md в system_prompt?
Сейчас agent.system_prompt — из БД (заполняется при создании агента). Если мы добавим agents.md — надо чтобы он был в system_prompt каждого агента.
Два варианта:
Вариант A: Копировать agents.md в каждого агента при создании (дублирование). Минус: обновлять в N местах.
Вариант B: Читать agents.md динамически при каждом запросе и добавлять в system_prompt (как сейчас с /mcp/list). Плюс: один файл для всех.
Мой выбор: Вариант B. В orchestrator.py — при формировании system_prompt — читать agents.md и добавлять в конец (как MCP-список).
Проблема 3: Где хранить agents.md?
Ты пишешь skills/agents/agents.md. Это папка с агентами. Логично — но есть нюанс:
•	Сейчас DATA_ROOT/skills/{AgentName}/ — папки агентов.
•	Если добавить DATA_ROOT/skills/agents/ — это будет папка с именем «agents» — как будто агент с именем «agents». Путаница.
Лучше: DATA_ROOT/agents.md (в корне DATA_ROOT), или DATA_ROOT/skills/_shared/agents.md.
Мой выбор: DATA_ROOT/agents.md (в корне). Просто, понятно, соответствует стандарту AGENTS.md.
Проблема 4: _base.md и agents.md — как объединить?
_base.md — конституция агента (общие правила поведения). agents.md — правила использования инструментов.

# AGENTS.md — Универсальные правила работы

## Как вызывать инструменты

Если пользователь просит **создать файл** — **ВСЕГДА** вызывай соответствующий инструмент. **НЕ ПЕРЕСКАЗЫВАЙ** содержимое, **НЕ ОСТАНАВЛИВАЙСЯ** на промежуточном шаге.

### Документ (DOCX)
- Пользователь: «Создай документ/отчёт»
- Вызывай: `create_docx` (**сразу**, без промежуточных шагов)

### Таблица (XLSX)
- Пользователь: «Создай таблицу»
- Вызывай: `create_xlsx`

### XML
- Пользователь: «Создай XML»
- Вызывай: `create_xml`

### Презентация (PPTX) — МНОГОШАГОВЫЙ
- Пользователь: «Сделай презентацию по файлу X»
- **ШАГ 1:** `read_file` (**получить** содержимое)
- **ШАГ 2:** `create_pptx` (**ОБЯЗАТЕЛЬНО**, **не** останавливайся после `read_file`)
- **ВАЖНО:** После `read_file` — **сразу** `create_pptx`. **НЕ** пересказывай содержимое. **НЕ** заканчивай работу.

### ТЗ из папки
- Пользователь: «Сделай ТЗ из папки X по шаблону Y»
- Вызывай: `create_doc_from_folder`

### Память
- «Запомни файл X» → `remember_file`
- «Что было в файле X?» / «Что ты запомнил?» → `recall_file`
- **НЕ ВЫЗЫВАЙ** `recall_file` на «Запомни» — это разные операции.

### Изучение методологии
- «Изучи ГОСТ/PMBOK/стандарт X» → `study_methodology`

## Правила поведения

1. **Всегда** возвращай JSON для инструментов — **без** текста до/после.
2. **Не** выдумывай инструменты — используй только из списка MCP.
3. **Не** пересказывай содержимое файла, если пользователь просит **создать** файл — вызывай `create_pptx` / `create_docx`.
4. Если инструмент вернул ошибку — попробуй **другой** подход (**не** повторяй тот же вызов).
5. Если не знаешь, что делать — **спроси** пользователя.

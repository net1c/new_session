Вот как они ложатся в наш проект:
1. mcp-bsl-platform-context — Справочник по платформе
•	Суть: MCP-сервер, который даёт агенту доступ к синтаксису, объектной модели и функциям 1С (как «Синтакс-помощник»).
•	Для нас: Это готовый инструмент (Tool) для агента-Разработчика. Он перестанет «гадать» и будет писать код, сверяясь с актуальной документацией платформы.
2. rlm-tools-bsl — Глаза и руки для анализа кода
•	Суть: Это RLM-сервер (Repository-Level Machine-coding) для точечного анализа больших конфигураций 1С (ERP, УТ). Он использует индексы и позволяет агенту находить конкретные процедуры, строить графы вызовов, искать ссылки, не читая тысячи строк кода.
•	Для нас: Это главный инструмент для агента-Аналитика и Разработчика. Вместо того чтобы грузить в LLM 2000 строк модуля, агент получит ровно нужную информацию, экономя токены и время. Это замена нашему RAG на старте — проще, быстрее и эффективнее, как показало сравнение (117K токенов против 265K у RAG).
3. ai_rules_1c — Методология и дисциплина
•	Суть: Это не инструмент, а набор правил, субагентов и слэш-команд для ИИ. Он структурирует работу агента: как писать код, как проводить ревью, как использовать MCP-инструменты, как работать с формами и расширениями. По сути — это готовый инженерный контекст.
•	Для нас: Мы перестаём писать системный промпт «с нуля». Внедряем эти правила как навык (Skill) для «Цифрового работника». Это превращает его из «болтуна» в профессионального 1С-разработчика, который знает стандарты и анти-паттерны.

def select_agent(self, item):
    text = item.text()
    try:
        # Извлекаем ID (оставляем как есть)
        agent_id = int(text.split("(id: ")[-1].rstrip(")"))
        # Извлекаем имя (всё, что до "(id: ")
        agent_name = text.split("(id: ")[0].strip()
        
        self.current_agent_id = agent_id
        self.current_agent_name = agent_name  # <--- СОХРАНЯЕМ ИМЯ
        self.chat_display.append(f"✅ Выбран агент: {text}")
    except Exception as e:
        self.chat_display.append(f"⚠️ Ошибка выбора агента: {e}")

def save_current_as_skill(self):
    """Сохраняет последний ответ агента как навык в его личную папку."""
    from PySide6.QtWidgets import QDialog, QVBoxLayout, QFormLayout, QLineEdit, QTextEdit, QDialogButtonBox
    import json
    from pathlib import Path
    from datetime import datetime
    import os

    self.chat_display.append("⏳ Подготовка к сохранению навыка...")
    QApplication.processEvents()

    # 1. Проверяем, есть ли активная задача и агент
    if not self.current_task_id:
        self.chat_display.append("⚠️ Нет активной задачи. Сначала выберите задачу.")
        return

    if not self.current_agent_name:
        self.chat_display.append("⚠️ Нет выбранного агента. Сначала выберите агента.")
        return

    # 2. Получаем историю и находим последнее сообщение агента
    try:
        response = requests.get(f"http://localhost:8000/history/{self.current_task_id}")
        if response.status_code != 200:
            self.chat_display.append(f"⚠️ Ошибка загрузки истории: {response.status_code}")
            return
        history = response.json()
        if not history:
            self.chat_display.append("⚠️ Нет сообщений в этой задаче.")
            return

        # Ищем последнее сообщение от агента
        last_agent_msg = None
        for msg in reversed(history):
            if msg["role"] == "agent":
                last_agent_msg = msg
                break

        if not last_agent_msg:
            self.chat_display.append("⚠️ Нет ответа агента для сохранения.")
            return

        agent_content = last_agent_msg["content"]
    except Exception as e:
        self.chat_display.append(f"❌ Ошибка при загрузке истории: {e}")
        return

    # 3. Открываем диалог для редактирования
    dialog = QDialog(self)
    dialog.setWindowTitle("Сохранить как навык")
    dialog.setMinimumWidth(500)
    layout = QVBoxLayout(dialog)

    form = QFormLayout()

    # Название навыка
    name_edit = QLineEdit()
    default_name = agent_content[:50].replace("\n", " ") + "..."
    name_edit.setText(default_name)
    form.addRow("📌 Название навыка:", name_edit)

    # Системный промпт
    prompt_edit = QTextEdit()
    prompt_edit.setPlainText(agent_content)
    prompt_edit.setPlaceholderText("Системный промпт для агента...")
    form.addRow("📝 Системный промпт:", prompt_edit)

    layout.addLayout(form)

    buttons = QDialogButtonBox(QDialogButtonBox.Ok | QDialogButtonBox.Cancel)
    buttons.accepted.connect(dialog.accept)
    buttons.rejected.connect(dialog.reject)
    layout.addWidget(buttons)

    if dialog.exec() != QDialog.Accepted:
        self.chat_display.append("❌ Сохранение навыка отменено.")
        return

    name = name_edit.text().strip()
    prompt = prompt_edit.toPlainText().strip()

    if not name or not prompt:
        self.chat_display.append("⚠️ Название и системный промпт не могут быть пустыми.")
        return

    # 4. Определяем путь для сохранения
    from config import SKILLS_ROOT
    skills_root = Path(SKILLS_ROOT)
    agent_folder = skills_root / self.current_agent_name
    agent_folder.mkdir(parents=True, exist_ok=True)  # Создаём папку, если её нет

    # Имя файла: название_навыка_дата_время.json
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    filename = f"{name.lower().replace(' ', '_')}_{timestamp}.json"
    filepath = agent_folder / filename

    # 5. Сохраняем навык в JSON
    skill_data = {
        "name": name,
        "system_prompt": prompt,
        "agent_name": self.current_agent_name,
        "saved_at": datetime.now().isoformat(),
        "source_task_id": self.current_task_id,
    }

    try:
        with open(filepath, "w", encoding="utf-8") as f:
            json.dump(skill_data, f, ensure_ascii=False, indent=2)
        self.chat_display.append(f"✅ Навык '{name}' сохранён в папку агента '{self.current_agent_name}'")
        self.chat_display.append(f"📁 {filepath}")
    except Exception as e:
        self.chat_display.append(f"❌ Ошибка при сохранении файла: {e}")

python
def _process_mcp_command(self, agent_response: str, time_str: str, agent_name: str) -> bool:
    """
    Проверяет, является ли ответ агента MCP-командой (JSON).
    Если да — выполняет команду и возвращает True (команда обработана).
    Если нет — возвращает False (показываем обычный ответ).
    """
    import json
    import re

    # --- ИЗВЛЕКАЕМ JSON ИЗ ТЕКСТА ---
    def extract_json_from_text(text: str) -> dict | None:
        # 1. Ищем JSON в кодовом блоке ```json ... ```
        code_block_match = re.search(r'```json\s*(\{.*?\})\s*```', text, re.DOTALL)
        if code_block_match:
            try:
                return json.loads(code_block_match.group(1))
            except json.JSONDecodeError:
                pass

        # 2. Ищем JSON с ключами 'tool' и 'params'
        json_match = re.search(r'\{[^{}]*"tool"\s*:\s*"[^"]+"\s*,\s*"params"\s*:\s*\{[^{}]*\}\s*\}', text, re.DOTALL)
        if json_match:
            try:
                return json.loads(json_match.group(0))
            except json.JSONDecodeError:
                pass

        # 3. Пробуем распарсить весь текст как JSON
        try:
            return json.loads(text.strip())
        except json.JSONDecodeError:
            pass

        return None
    # ----------------------------------

    tool_call = extract_json_from_text(agent_response)

    if not tool_call:
        return False  # Не удалось найти JSON

    if not isinstance(tool_call, dict) or "tool" not in tool_call:
        return False

    # Если это MCP-команда — отправляем на выполнение
    self.chat_display.append(f"🔧 [{time_str}] {agent_name} вызывает инструмент: {tool_call['tool']}")
    from PySide6.QtWidgets import QApplication
    QApplication.processEvents()

    try:
        exec_response = requests.post(
            "http://localhost:8000/mcp/execute",
            json=tool_call,
            timeout=60,
        )

        if exec_response.status_code == 200:
            exec_data = exec_response.json()
            if exec_data.get("status") == "ok":
                self.chat_display.append(f"✅ [{time_str}] Файл создан: {exec_data.get('filepath')}")
                self.chat_display.append(f"📝 {exec_data.get('message', '')}")
            else:
                self.chat_display.append(f"⚠️ [{time_str}] Ошибка при создании файла: {exec_data.get('message')}")
        else:
            self.chat_display.append(f"⚠️ [{time_str}] Ошибка при вызове инструмента: {exec_response.status_code}")

        return True  # Команда обработана

    except Exception as e:
        self.chat_display.append(f"❌ Ошибка при обработке MCP-команды: {e}")
        return True
